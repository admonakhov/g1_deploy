cmake_minimum_required(VERSION 3.16)
project(fast_teleop_g1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(TELEOP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(UNITREE_DEPLOY_SUPPORT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/unitree_deploy/include")
unset(UNITREE_RL_ROOT CACHE)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/OnnxRuntime.cmake")
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
add_library(teleop_core INTERFACE)
target_include_directories(teleop_core INTERFACE include)
target_link_libraries(teleop_core INTERFACE Eigen3::Eigen yaml-cpp::yaml-cpp nlohmann_json::nlohmann_json Threads::Threads teleop_onnxruntime)
target_compile_options(teleop_core INTERFACE -Wall -Wextra -Wpedantic)
target_compile_definitions(teleop_core INTERFACE TELEOP_ROOT="${TELEOP_ROOT}")
add_executable(g1_stream_preview src/preview.cpp)
target_link_libraries(g1_stream_preview PRIVATE teleop_core)
add_executable(g1_policy_check src/policy_check.cpp)
target_link_libraries(g1_policy_check PRIVATE teleop_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_teleop tests/test_core.cpp)
  target_link_libraries(test_teleop PRIVATE teleop_core)
  add_test(NAME teleop_core COMMAND test_teleop)
  add_executable(g1_policy_fixture tests/policy_fixture.cpp)
  target_link_libraries(g1_policy_fixture PRIVATE teleop_core)
endif()
option(BUILD_ROBOT_CONTROLLER "Build Unitree DDS / FSM controller" ON)
if(BUILD_ROBOT_CONTROLLER)
  find_package(unitree_sdk2 REQUIRED)
  find_package(Boost CONFIG REQUIRED COMPONENTS program_options)
  find_package(spdlog REQUIRED)
  find_package(fmt REQUIRED)
  add_executable(g1_ctrl src/main.cpp src/State_Policy.cpp)
  target_include_directories(g1_ctrl SYSTEM PRIVATE "${UNITREE_DEPLOY_SUPPORT}")
  target_link_libraries(g1_ctrl PRIVATE teleop_core unitree_sdk2 Boost::program_options spdlog::spdlog_header_only fmt::fmt-header-only rt)
  if(BUILD_TESTING)
    add_executable(g1_fsm_keys_fixture tests/fsm_keys_fixture.cpp)
    target_include_directories(g1_fsm_keys_fixture SYSTEM PRIVATE "${UNITREE_DEPLOY_SUPPORT}")
    target_link_libraries(g1_fsm_keys_fixture PRIVATE teleop_core unitree_sdk2)
  endif()
endif()
